# ── CPU-ONLY PyTorch Installation ──────────────────────────────────────────────
# CRITICAL: Use --index-url (NOT --extra-index-url) to make the CPU wheel
# repository the PRIMARY source. PyPI is added back as secondary via
# --extra-index-url. This guarantees pip resolves torch from the CPU index
# first and NEVER downloads GPU/CUDA/NVIDIA wheels.
#
# The +cpu local version suffix ensures pip cannot accidentally match
# a GPU wheel from PyPI (PyPI has torch==2.2.2, NOT torch==2.2.2+cpu).
# ───────────────────────────────────────────────────────────────────────────────
--index-url https://download.pytorch.org/whl/cpu
--extra-index-url https://pypi.org/simple/

torch==2.12.0+cpu
sentence-transformers==2.7.0
flask
flask-cors
pymupdf

chromadb==1.5.9

# ── LangChain ─────────────────────────────────────────────────────────────────
# The bare `langchain` umbrella package is deliberately NOT included: an LCEL
# chain (prompt | llm) needs only langchain-core, and the umbrella adds weight
# this free-tier instance cannot spare.
#
# langchain-huggingface is pinned to 1.0.0, NOT latest. 1.2.2 declares
# sentence-transformers>=5.2.0 / transformers>=5.0.0, which would force a major
# upgrade of the embedding stack out from under the frozen baseline in
# validation/. 1.0.0 declares sentence-transformers>=2.6.0,<3.0.0 and
# transformers>=4.39.0,<5.0.0 — an exact fit for the pins above.
langchain-core==1.5.3
langchain-chroma==1.1.0
langchain-groq==1.1.3
langchain-huggingface==1.0.0
