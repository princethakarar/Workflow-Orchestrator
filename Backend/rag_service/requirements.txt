# ── CPU-only, memory-constrained deployment ────────────────────────────────────
# This service runs on a Render free instance capped at 512 MB RSS.
#
# torch and sentence-transformers were REMOVED deliberately. They cost ~364 MB
# of the budget (torch import 171 MB, MiniLM weights 124 MB, inference spike
# 69 MB) and pushed the service to a ~559 MB peak, which is the OOM. Embeddings
# now run on ONNX Runtime, which ships as a chromadb dependency, using the same
# all-MiniLM-L6-v2 model. Verified equivalent: cosine similarity 1.000000
# against the frozen torch vectors in validation/golden_chunks_v1.json.
#
# The pytorch CPU wheel index that used to be pinned here is gone with torch —
# everything below resolves from PyPI.
# ───────────────────────────────────────────────────────────────────────────────

flask
flask-cors
pymupdf

# Pinned: the on-disk chroma_db/ store was written under 1.5.9's schema, and
# langchain-chroma only constrains it to >=1.3.5,<2.0.0 — wide enough to drift
# into a schema change on a rebuild. Also supplies onnxruntime + the ONNX
# MiniLM embedding function the service now depends on.
chromadb==1.5.9

# Production WSGI server (see Dockerfile CMD).
gunicorn

# ── LangChain ─────────────────────────────────────────────────────────────────
# The bare `langchain` umbrella package is deliberately NOT included: an LCEL
# chain (prompt | llm) needs only langchain-core.
#
# langchain-huggingface is likewise gone — it exists to wrap
# sentence-transformers, which is no longer installed. The ONNX embeddings are
# exposed to LangChain through a small Embeddings subclass in app.py.
langchain-core==1.5.3
langchain-chroma==1.1.0
langchain-groq==1.1.3

# Local-dev config loading for the RAG service (Render sets env vars directly).
python-dotenv
